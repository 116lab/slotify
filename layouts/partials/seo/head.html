{{- /* seo/head.html Single source of truth for per-page crawler/social meta. Called by layouts/_default/baseof.html for EVERY HTML page. Branches on the page Kind so one partial serves home, single posts, sections, and taxonomy/term lists. Why this matters: the site is a client-rendered SPA, but social crawlers (Facebook/Line/Twitter/Slack) and many search bots do NOT execute JavaScript. Whatever share preview or snippet they show comes ENTIRELY from this server-rendered block — never from the SPA's runtime document.title updates. Context (.) is the page. */ -}} {{- $site := .Site -}} {{- $siteTitle := $site.Title -}} {{- $author := .Params.author | default $site.Params.author | default "" -}} {{- /* Title: "Page Title | Site Title", but the home page is just the site title. */ -}} {{- $title := cond .IsHome $siteTitle (printf "%s | %s" .Title $siteTitle) -}} {{- /* Description: pages prefer summary/description then the auto excerpt; list and home pages fall back to the site tagline. Collapsed to one clean line. */ -}} {{- $desc := "" -}} {{- if .IsPage -}} {{- $desc = or .Params.description .Params.summary .Summary -}} {{- else -}} {{- $desc = or .Params.description $site.Params.description -}} {{- end -}} {{- /* plainify strips tags but leaves HTML entities (e.g. ’); htmlUnescape turns them back into plain characters so the value is encoded exactly once when emitted into an attribute / JSON-LD string (no &rsquo; doubling). */ -}} {{- $desc = $desc | plainify | htmlUnescape | replaceRE "\\s+" " " | strings.TrimSpace -}} {{- /* Share image: a page bundle/front-matter image, else the site default. absURL leaves an already-absolute external URL untouched and makes a root-relative bundle path fully qualified (required by OG/Twitter). */ -}} {{- $img := partial "func/resolve-image.html" . -}} {{- $img = or $img $site.Params.ogImage (index ($site.Params.images | default slice) 0) -}} {{- with $img }}{{ $img = . | absURL }}{{ end -}} {{- /* article for a single post/page, website for home/list/taxonomy. */ -}} {{- $ogType := cond .IsPage "article" "website" -}} {{ $title }} {{- with $author }}{{ end }} {{- with $img }} {{- end }} {{- if .IsPage }} {{- if not .Lastmod.IsZero }} {{- end }} {{- with $author }}{{ end }} {{- range .Param "tags" }} {{- end }} {{- end }} {{- with $img }} {{- end }} {{- /* JSON-LD structured data: BlogPosting for a single article, WebSite for everything else. Built as a dict + jsonify so quoting/escaping is always valid regardless of title/description content. */ -}} {{- $ld := dict -}} {{- if .IsPage -}} {{- $ld = dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" .Title "description" $desc "datePublished" (.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" (.Lastmod.Format "2006-01-02T15:04:05Z07:00") "url" .Permalink "mainEntityOfPage" .Permalink "wordCount" .WordCount "author" (dict "@type" "Person" "name" $author) "publisher" (dict "@type" "Organization" "name" $siteTitle) -}} {{- with $img }}{{ $ld = merge $ld (dict "image" .) }}{{ end -}} {{- else -}} {{- $ld = dict "@context" "https://schema.org" "@type" "WebSite" "name" $siteTitle "description" $desc "url" .Permalink -}} {{- end -}}
